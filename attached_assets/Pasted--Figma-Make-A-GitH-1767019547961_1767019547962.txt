ПРИМЕР CONTAINER КОМПОНЕНТА (создай ExampleContainer.tsx):

```typescript
/**
 * 📦 CONTAINER PATTERN
 * 
 * Container компонент содержит:
 * - Логику (useState, useEffect, API вызовы)
 * - Бизнес-логику (расчёты, форматирование)
 * - Supabase запросы
 * 
 * И импортирует UI компонент из /figma-ui
 */

import { useState, useEffect } from 'react';
import { CEOMissionControlView } from '../../figma-ui/components/dashboard/CEOMissionControlView';
import { useDashboardData } from '../../hooks/useDashboardData';
import * as api from '../../utils/api';

export function CEOMissionControlContainer({ currentUser }: { currentUser: any }) {
  const [loading, setLoading] = useState(true);
  const [stats, setStats] = useState(null);
  
  // ВСЯ ЛОГИКА ЗДЕСЬ
  useEffect(() => {
    loadData();
  }, []);
  
  const loadData = async () => {
    try {
      const response = await api.getAdminStats();
      setStats(response.stats);
    } catch (error) {
      console.error(error);
    } finally {
      setLoading(false);
    }
  };
  
  // ПРОСТО ПЕРЕДАЁМ ПРОПСЫ В UI
  return (
    <CEOMissionControlView 
      loading={loading}
      stats={stats}
      onRefresh={loadData}
    />
  );
}
